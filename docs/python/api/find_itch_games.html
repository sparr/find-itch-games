find_itch_games

Find the itch.io app, its install locations, and the games installed in them.

A Python counterpart to the find-itch-games JavaScript package. Both read the same two records itch keeps -- butler.db and the receipts itch writes into install folders -- and are tested against the same shared fixtures.

Start with find_itch() for the whole picture, or find_itch_app_by_id() / find_itch_app_by_name() to locate one game.

117#: Which of itch's records to trust.
118#:
119#: - ``merge`` (default) reads butler.db and confirms each game against the
120#:   receipt in its install folder, then adds any receipt the database has no
121#:   row for. Most complete.
122#: - ``db`` reads butler.db only. Fastest, and the only source of play times,
123#:   launch candidates and custom install folders.
124#: - ``receipts`` scans the install locations instead of reading caves. Useful
125#:   when the database is out of date, but it loses the fields only the
126#:   database has. Install locations still come from the database, since
127#:   nothing on disk records them.
128Strategy = Literal["merge", "db", "receipts"]
129
130#: Every value :data:`Strategy` accepts, in preference order.
131#:
132#: A tuple because it is the guard, not a registry: :func:`find_itch` dispatches
133#: on these values directly, so widening the list would only disable the check
134#: and let the new value fall through to ``merge`` behaviour.
135ITCH_STRATEGIES: tuple[str, ...] = ("merge", "db", "receipts")

168def _resolve_strategy(strategy: str | None, ignore_database: bool) -> str:
169    """Settles on a strategy, rejecting anything that isn't one.
170
171    Every strategy is defined by what it is *not*, so a misspelled value would
172    otherwise slip through and behave like ``merge``.
173    """
174    # Validated before `ignore_database` is honoured, so that argument cannot
175    # mask a typo in a value the caller clearly meant to have an effect.
176    if strategy is not None and strategy not in ITCH_STRATEGIES:
177        expected = ", ".join(repr(s) for s in ITCH_STRATEGIES)
178        raise ValueError(f"unknown itch strategy {strategy!r}; expected one of {expected}")
179    if ignore_database:
180        return "receipts"
181    return strategy or "merge"

184class _LibrarySet:
185    """Groups apps under the install location containing them, synthesizing an
186    entry for anything installed outside a known one."""
187
188    def __init__(self, libraries: list[LibraryRaw]) -> None:
189        self._by_path: dict[str, Library] = {
190            str(library.path): Library(**vars(library), apps=[]) for library in libraries
191        }
192        self._seen: set[str] = set()
193
194    @property
195    def existing(self) -> list[Library]:
196        return [library for library in self._by_path.values() if library.exists]
197
198    def find_by_id(self, location_id: str) -> Library | None:
199        return next((l for l in self._by_path.values() if l.id == location_id), None)
200
201    def has_app(self, app_path: Path) -> bool:
202        return str(app_path.resolve()) in self._seen
203
204    def add_app(self, app: App) -> None:
205        resolved = app.path.resolve()
206        if str(resolved) in self._seen:
207            return
208        self._seen.add(str(resolved))
209
210        parent = str(resolved.parent)
211        library = self._by_path.get(parent)
212        if library is None:
213            # A custom install folder, or a location itch has since forgotten.
214            library = Library(
215                id=app.manifest.install_location_id or f"external:{parent}",
216                path=resolved.parent,
217                is_default=False,
218                exists=True,
219                source="option",
220                apps=[],
221            )
222            self._by_path[parent] = library
223        library.apps.append(app)
224
225    def to_list(self) -> list[Library]:
226        for library in self._by_path.values():
227            library.apps.sort(key=lambda app: app.game_id)
228        return sorted(self._by_path.values(), key=lambda library: str(library.path))

231def _resolve_cave_install_path(cave: CaveRow, libraries: _LibrarySet) -> Path | None:
232    """Works out where a cave's files are.
233
234    ``custom_install_folder`` wins, then the install location plus folder name,
235    and finally the base path butler recorded when it last scanned the folder --
236    the only hint left if the install location has been removed.
237    """
238    if cave.custom_install_folder:
239        return Path(cave.custom_install_folder).resolve()
240
241    library = libraries.find_by_id(cave.install_location_id) if cave.install_location_id else None
242    if library is not None and cave.install_folder_name:
243        return get_install_folder_path(library.path, cave.install_folder_name)
244
245    if cave.verdict and cave.verdict.base_path:
246        return Path(cave.verdict.base_path).resolve()
247    return None

250def find_itch(
251    *,
252    itch_path: Path | str | None = None,
253    strategy: Strategy | None = None,
254    check_exists: bool = True,
255    extra_libraries: list[Path | str] | None = None,
256    ignore_database: bool = False,
257    lookup: PathLookup | None = None,
258) -> Libraries:
259    """Finds itch, its install locations and every game installed in them.
260
261    :raises ItchNotFoundError: if the itch user-data directory cannot be found.
262    :raises ValueError: if ``strategy`` is not one of :data:`ITCH_STRATEGIES`.
263    """
264    resolved_strategy = _resolve_strategy(strategy, ignore_database)
265    itch_dir = require_itch_path(itch_path, lookup)
266
267    raw = load_itch_libraries(
268        itch_dir, extra_libraries=extra_libraries, skip_database=ignore_database
269    )
270    libraries = _LibrarySet(raw.libraries)
271
272    if resolved_strategy != "receipts" and raw.database_available:
273        with open_butler_database(raw.database_path) as connection:
274            caves = read_caves(connection)
275        for cave in caves:
276            install_path = _resolve_cave_install_path(cave, libraries)
277            if install_path is None:
278                continue
279            if check_exists and not install_path.is_dir():
280                continue
281            receipt = None if resolved_strategy == "db" else read_receipt(install_path)
282            manifest = manifest_from_cave(cave, install_path, receipt)
283            libraries.add_app(
284                App(
285                    game_id=manifest.game_id,
286                    path=manifest.path,
287                    manifest=manifest,
288                    receipt_path=manifest.receipt_path,
289                )
290            )
291
292    if resolved_strategy != "db":
293        for library in libraries.existing:
294            for found in find_receipts(library.path):
295                if libraries.has_app(found.path):
296                    continue
297                manifest = manifest_from_receipt(found, library.id)
298                libraries.add_app(
299                    App(
300                        game_id=manifest.game_id,
301                        path=manifest.path,
302                        manifest=manifest,
303                        receipt_path=manifest.receipt_path,
304                    )
305                )
306
307    return Libraries(
308        itch_path=raw.itch_path,
309        database_path=raw.database_path,
310        database_available=raw.database_available,
311        strategy=resolved_strategy,
312        libraries=libraries.to_list(),
313    )

342def _pick_one(apps: list[App], query: str | int, strict: bool) -> App:
343    if not apps:
344        raise AppNotFoundError(query)
345    if strict and len(apps) > 1:
346        raise AmbiguousAppError(query, [str(app.path) for app in apps])
347    return apps[0]

350def find_itch_apps_by_id(game_id: int, **options) -> list[App]:
351    """Every install of a game, by itch.io game id.
352
353    Usually zero or one, but itch will happily install the same game into more
354    than one location.
355    """
356    return [app for app in find_itch_apps(**options) if app.game_id == game_id]

359def find_itch_app_by_id(game_id: int, *, strict: bool = False, **options) -> Path:
360    """Finds where a game is installed, by itch.io game id.
361
362    :raises AppNotFoundError: if the game is not installed.
363    :raises AmbiguousAppError: if ``strict`` and it is installed twice.

368def find_itch_apps_by_name(name: str, *, exact: bool = True, **options) -> list[App]:
369    """Every installed game matching a title, url slug or install folder name.
370
371    Exact by default; pass ``exact=False`` for a search that ignores case,
372    spacing and punctuation.
373    """
374    wanted = name if exact else normalize_name(name)
375    return [
376        app
377        for app in find_itch_apps(**options)
378        if any(
379            (alias == wanted) if exact else (normalize_name(alias) == wanted)
380            for alias in get_manifest_names(app.manifest)
381        )
382    ]

385def find_itch_app_by_name(
386    name: str, *, exact: bool = True, strict: bool = False, **options
387) -> Path:
388    """Finds where a game is installed, by title, url slug or install folder name.
389
390    :raises AppNotFoundError: if no installed game matches.
391    :raises AmbiguousAppError: if ``strict`` and several match.
392    """
393    return _pick_one(find_itch_apps_by_name(name, exact=exact, **options), name, strict).path

402def has_itch_app(game: int | str, *, exact: bool = True, **options) -> bool:
403    """Whether a game is currently installed, by game id or by name."""
404    if isinstance(game, int):
405        return bool(find_itch_apps_by_id(game, **options))
406    return bool(find_itch_apps_by_name(game, exact=exact, **options))

def get_itch_path_candidates( lookup: PathLookup | None = None) -> list[pathlib.Path]: 
 67def get_itch_path_candidates(lookup: PathLookup | None = None) -> list[Path]:
 68    """Directories itch may keep its user data in, most specific first.
 69
 70    Mirrors Electron's ``app.getPath("userData")`` for each app variant.
 71
 72    ``ITCH_USER_DATA_DIR`` and ``ITCH_APP_DIR`` are this library's own escape
 73    hatch for unusual installs -- the itch app itself does not read them.
 74    """
 75    home, env, platform = (lookup or PathLookup()).resolved()
 76    candidates: list[Path] = []
 77
 78    def push(*parts: str | Path | None) -> None:
 79        if all(parts):
 80            candidates.append(Path(*(str(p) for p in parts)))  # type: ignore[arg-type]
 81
 82    push(env.get("ITCH_USER_DATA_DIR"))
 83    push(env.get("ITCH_APP_DIR"))
 84
 85    for app_name in ITCH_APP_NAMES:
 86        if platform == "win32":
 87            # Electron's `appData` on Windows is %APPDATA% (Roaming).
 88            push(env.get("APPDATA"), app_name)
 89            push(home, "AppData", "Roaming", app_name)
 90            push(env.get("LOCALAPPDATA"), app_name)
 91        elif platform == "darwin":
 92            push(home, "Library", "Application Support", app_name)
 93        else:
 94            push(env.get("XDG_CONFIG_HOME"), app_name)
 95            push(home, ".config", app_name)
 96            # Flatpak redirects XDG_CONFIG_HOME inside the sandbox only; from
 97            # outside it, the app's config lives here.
 98            push(home, ".var", "app", f"io.{app_name}.{app_name}", "config", app_name)
 99
100    seen: dict[str, Path] = {}
101    for candidate in candidates:
102        seen.setdefault(str(candidate), candidate)
103    return list(seen.values())
